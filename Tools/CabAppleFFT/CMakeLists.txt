cmake_minimum_required(VERSION 3.22)
project(ChimeraCabAppleFFTResearch LANGUAGES C CXX)
if(APPLE)
  enable_language(OBJC OBJCXX)
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(product "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
if(NOT JUCE_UPSTREAM)
  message(FATAL_ERROR "Pass -DJUCE_UPSTREAM=/path/to/clean/JUCE-8.0.8")
endif()
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate.py"
  --juce "${JUCE_UPSTREAM}" --output "${CMAKE_BINARY_DIR}/overlay" COMMAND_ERROR_IS_FATAL ANY)
add_subdirectory("${CMAKE_BINARY_DIR}/overlay/JUCE" "${CMAKE_BINARY_DIR}/juce-build" EXCLUDE_FROM_ALL)
juce_add_binary_data(AppleResearchIRs NAMESPACE ChimeraIRData HEADER_NAME ChimeraIRData.h
  SOURCES "${product}/Assets/IRs/guitar_v30_sm57.wav" "${product}/Assets/IRs/guitar_jensen_sm57.wav")
add_library(AppleResearchJuce STATIC)
target_link_libraries(AppleResearchJuce PRIVATE juce::juce_data_structures juce::juce_dsp
  juce::juce_audio_formats juce::juce_cryptography)
target_include_directories(AppleResearchJuce PUBLIC "${CMAKE_BINARY_DIR}/overlay/JUCE/modules")
target_compile_definitions(AppleResearchJuce PUBLIC JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1
  JUCE_STANDALONE_APPLICATION=1 JUCE_USE_CURL=0 JUCE_WEB_BROWSER=0)
foreach(variant baseline candidate)
  set(overlay "${CMAKE_BINARY_DIR}/overlay/${variant}/Source")
  foreach(kind fft-check fft-probe parity realtime integration ir-state probe)
    set(target "apple-${variant}-${kind}")
    add_executable(${target})
    if(kind STREQUAL fft-check)
      target_sources(${target} PRIVATE FFTCheck.cpp)
    elseif(kind STREQUAL fft-probe)
      target_sources(${target} PRIVATE FFTProbe.cpp)
    elseif(kind STREQUAL parity)
      target_sources(${target} PRIVATE "${product}/Tools/CabZeroWork/Parity.cpp")
    elseif(kind STREQUAL realtime)
      target_sources(${target} PRIVATE "${product}/Tests/CabRealtimeRegressionTests.cpp")
    elseif(kind STREQUAL integration)
      target_sources(${target} PRIVATE "${product}/Tests/OriginalCabIntegrationTests.cpp" "${overlay}/IRLibrary.cpp")
    elseif(kind STREQUAL ir-state)
      target_sources(${target} PRIVATE "${product}/Tests/IRLibraryTests.cpp" "${overlay}/IRLibrary.cpp")
    else()
      target_sources(${target} PRIVATE "${product}/Tools/CabZeroWork/Probe.cpp")
    endif()
    if(variant STREQUAL candidate)
      target_compile_definitions(${target} PRIVATE CAB_APPLE_CANDIDATE=1)
    endif()
    target_include_directories(${target} PRIVATE "${overlay}" "${product}/Source" "${product}/Tests")
    target_compile_definitions(${target} PRIVATE JUCE_USE_CURL=0 JUCE_WEB_BROWSER=0)
    target_link_libraries(${target} PRIVATE AppleResearchIRs AppleResearchJuce)
    # Give the collector stable paths.
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  endforeach()
endforeach()

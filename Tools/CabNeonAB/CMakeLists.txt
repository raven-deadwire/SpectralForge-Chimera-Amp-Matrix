cmake_minimum_required(VERSION 3.22)
project(ChimeraCabNeonAB VERSION 0.1.0 LANGUAGES C CXX)
if(APPLE)
  enable_language(OBJCXX)
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CHIMERA_PRODUCT_SOURCE OR NOT CHIMERA_PROBE_SOURCE)
  message(FATAL_ERROR "Exact product source and generated diagnostic harness are required")
endif()
enable_testing()
add_subdirectory("${CHIMERA_PRODUCT_SOURCE}" product EXCLUDE_FROM_ALL)
foreach(target IN ITEMS ChimeraCabABControl ChimeraCabABProfile)
  juce_add_console_app(${target} PRODUCT_NAME "${target}")
  target_sources(${target} PRIVATE "${CHIMERA_PROBE_SOURCE}" "${CHIMERA_PRODUCT_SOURCE}/Source/IRLibrary.cpp")
  target_include_directories(${target} PRIVATE "${CHIMERA_PRODUCT_SOURCE}/Source" "${CHIMERA_PRODUCT_SOURCE}/Tests")
  target_compile_definitions(${target} PRIVATE JUCE_USE_CURL=0 JUCE_WEB_BROWSER=0)
  target_link_libraries(${target} PRIVATE ChimeraIRs juce::juce_data_structures juce::juce_dsp juce::juce_audio_formats PUBLIC juce::juce_recommended_config_flags)
endforeach()
target_compile_definitions(ChimeraCabABProfile PRIVATE CHIMERA_CAB_PROFILE=1)

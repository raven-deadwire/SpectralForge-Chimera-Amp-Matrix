cmake_minimum_required(VERSION 3.22)
project(CabFFTSharing LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(APPLE)
  enable_language(OBJCXX)
endif()
if(NOT VARIANTS OR NOT JUCE_SOURCE)
  message(FATAL_ERROR "Explicit generated VARIANTS and pinned JUCE_SOURCE required")
endif()
set(JUCE_MODULES_DIR "${JUCE_SOURCE}/modules")
include("${JUCE_SOURCE}/extras/Build/CMake/JUCEModuleSupport.cmake")
juce_add_modules(ALIAS_NAMESPACE juce "${JUCE_SOURCE}/modules/juce_core" "${JUCE_SOURCE}/modules/juce_audio_basics" "${JUCE_SOURCE}/modules/juce_audio_formats" "${JUCE_SOURCE}/modules/juce_dsp")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/empty.cpp" "// Shared JUCE translation units.\n")
add_library(cab_runtime STATIC "${CMAKE_CURRENT_BINARY_DIR}/empty.cpp")
target_link_libraries(cab_runtime PRIVATE juce::juce_dsp)
target_include_directories(cab_runtime PUBLIC "${JUCE_SOURCE}/modules")
target_compile_definitions(cab_runtime PUBLIC JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1 JUCE_STANDALONE_APPLICATION=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_MODAL_LOOPS_PERMITTED=0)
enable_testing()
foreach(variant IN ITEMS A B)
  foreach(mode IN ITEMS control profile)
    add_executable(${variant}_${mode} probe.cpp)
    target_include_directories(${variant}_${mode} PRIVATE "${VARIANTS}/${variant}/Source")
    target_link_libraries(${variant}_${mode} PRIVATE cab_runtime)
    target_compile_definitions(${variant}_${mode} PRIVATE EXPERIMENT_CANDIDATE=$<STREQUAL:${variant},B>)
    if(mode STREQUAL "profile")
      target_compile_definitions(${variant}_${mode} PRIVATE CHIMERA_CAB_PROFILE=1)
    endif()
  endforeach()
  add_executable(${variant}_original_contracts "${VARIANTS}/${variant}/Tests/CabRealtimeRegressionTests.cpp")
  target_include_directories(${variant}_original_contracts PRIVATE "${VARIANTS}/${variant}/Source")
  target_link_libraries(${variant}_original_contracts PRIVATE cab_runtime)
  add_test(NAME ${variant}_original_contracts COMMAND ${variant}_original_contracts)
endforeach()

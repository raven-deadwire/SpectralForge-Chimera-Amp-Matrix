cmake_minimum_required(VERSION 3.22)
project(ChimeraCabZeroWork LANGUAGES C CXX)
if(APPLE)
  enable_language(OBJCXX)
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(product "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
# Existing product build supplies pinned JUCE and binary IR targets. No change
# to its DSP, platform flags, original acceptance targets or release policy.
add_subdirectory("${product}" "${CMAKE_BINARY_DIR}/product" EXCLUDE_FROM_ALL)
enable_testing()
foreach(variant baseline empty_partition zero_history)
  set(overlay "${CMAKE_BINARY_DIR}/${variant}")
  file(MAKE_DIRECTORY "${overlay}")
  file(COPY "${product}/Source" DESTINATION "${overlay}")
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate.py"
    --source "${product}/Source/ModeledCabConvolution.h" --candidate "${variant}"
    --output "${overlay}/Source/ModeledCabConvolution.h" COMMAND_ERROR_IS_FATAL ANY)
  file(READ "${product}/Source/ModeledCabConvolution.h" original)
  string(REPLACE "ModeledCabConvolution" "BaselineConvolution" reference "${original}")
  file(WRITE "${overlay}/Source/BaselineConvolution.h" "${reference}")
  foreach(kind probe parity realtime integration ir-state)
    set(target "zero-${variant}-${kind}")
    juce_add_console_app(${target} PRODUCT_NAME "${target}")
    if(kind STREQUAL probe)
      target_sources(${target} PRIVATE Probe.cpp)
    elseif(kind STREQUAL parity)
      target_sources(${target} PRIVATE Parity.cpp)
    elseif(kind STREQUAL realtime)
      target_sources(${target} PRIVATE "${product}/Tests/CabRealtimeRegressionTests.cpp")
    elseif(kind STREQUAL integration)
      target_sources(${target} PRIVATE "${product}/Tests/OriginalCabIntegrationTests.cpp" "${overlay}/Source/IRLibrary.cpp")
    else()
      target_sources(${target} PRIVATE "${product}/Tests/IRLibraryTests.cpp" "${overlay}/Source/IRLibrary.cpp")
    endif()
    target_include_directories(${target} PRIVATE "${overlay}/Source" "${product}/Tests")
    target_compile_definitions(${target} PRIVATE JUCE_USE_CURL=0 JUCE_WEB_BROWSER=0)
    target_link_libraries(${target} PRIVATE ChimeraIRs juce::juce_data_structures juce::juce_dsp
      juce::juce_audio_formats juce::juce_cryptography PUBLIC juce::juce_recommended_config_flags)
  endforeach()
endforeach()

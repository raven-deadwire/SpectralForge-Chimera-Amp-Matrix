cmake_minimum_required(VERSION 3.22)
project(ChimeraTransposeLab VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
# Explicit, local dependencies keep the research build out of release CI.
set(JUCE_SOURCE "" CACHE PATH "JUCE 8.0.8 source checkout")
set(STRETCH_SOURCE "" CACHE PATH "Signalsmith Stretch pinned checkout")
set(LINEAR_SOURCE "" CACHE PATH "Signalsmith Linear 0.6.4 checkout")
if(NOT EXISTS "${JUCE_SOURCE}/CMakeLists.txt")
  message(FATAL_ERROR "Set JUCE_SOURCE to a JUCE 8.0.8 checkout")
endif()
add_subdirectory("${JUCE_SOURCE}" juce)
get_filename_component(CHIMERA_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
enable_testing()
foreach(name TransposeLab PitchRegressionTests StabilityTests GateTests)
  juce_add_console_app(${name} PRODUCT_NAME "${name}")
  if(name STREQUAL "TransposeLab")
    target_sources(${name} PRIVATE TransposeLab.cpp)
  else()
    target_sources(${name} PRIVATE "${CHIMERA_ROOT}/Tests/${name}.cpp")
    add_test(NAME ${name} COMMAND ${name})
  endif()
  target_include_directories(${name} PRIVATE "${CHIMERA_ROOT}/Source")
  target_compile_definitions(${name} PRIVATE JUCE_USE_CURL=0 JUCE_WEB_BROWSER=0)
  target_link_libraries(${name} PRIVATE juce::juce_dsp juce::juce_audio_basics
    PUBLIC juce::juce_recommended_config_flags)
endforeach()
if(EXISTS "${STRETCH_SOURCE}/signalsmith-stretch.h" AND EXISTS "${LINEAR_SOURCE}/stft.h")
  # The header includes signalsmith-linear/stft.h. Use the upstream include tree.
  target_include_directories(TransposeLab PRIVATE "${STRETCH_SOURCE}" "${LINEAR_SOURCE}/include")
  target_compile_definitions(TransposeLab PRIVATE CHIMERA_LAB_SIGNALSMITH=1)
endif()

cmake_minimum_required(VERSION 3.22)
project(ChimeraNamTools VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
if(NOT TARGET juce::juce_dsp)
  if(NOT CHIMERA_JUCE_SOURCE)
    message(FATAL_ERROR "Set CHIMERA_JUCE_SOURCE to JUCE 8.0.8")
  endif()
  add_subdirectory("${CHIMERA_JUCE_SOURCE}" juce EXCLUDE_FROM_ALL)
endif()

if(CHIMERA_TONE3000_SOURCE)
  set(T3K "${CHIMERA_TONE3000_SOURCE}/plugin")
  set(T3K_NAM "${T3K}/NeuralAmpModelerCore")
  set(T3K_NAM_SOURCES)
  foreach(part activations conv1d container convnet dsp get_dsp lstm ring_buffer util wavenet/a2_fast wavenet/model wavenet/slimmable)
    list(APPEND T3K_NAM_SOURCES "${T3K_NAM}/NAM/${part}.cpp")
  endforeach()
  juce_add_console_app(ChimeraTone3000Tool PRODUCT_NAME "Chimera TONE3000 Validation")
  target_sources(ChimeraTone3000Tool PRIVATE Tone3000Tool.cpp
    "${T3K}/src/NamEngine.cpp" "${T3K}/src/BlockEq.cpp" "${T3K}/src/PresetFile.cpp" ${T3K_NAM_SOURCES})
  target_include_directories(ChimeraTone3000Tool PRIVATE "${T3K}/include" "${T3K}/AudioDSPTools"
    "${T3K_NAM}" "${T3K_NAM}/Dependencies/eigen" "${T3K_NAM}/Dependencies/nlohmann")
  target_compile_definitions(ChimeraTone3000Tool PRIVATE JUCE_USE_CURL=0 JUCE_WEB_BROWSER=0 NAM_ENABLE_A2_FAST=1 EIGEN_MPL2_ONLY=1 EIGEN_NO_DEBUG=1)
  target_link_libraries(ChimeraTone3000Tool PRIVATE juce::juce_audio_basics juce::juce_data_structures juce::juce_graphics PUBLIC juce::juce_recommended_config_flags)
endif()

cmake_minimum_required(VERSION 3.20)
project(ChimeraPerformanceCollectorContracts LANGUAGES CXX)
add_executable(ChimeraPerformanceCollectorTest test_recorder.cpp)
target_compile_features(ChimeraPerformanceCollectorTest PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(ChimeraPerformanceCollectorTest PRIVATE /W4)
else()
  target_compile_options(ChimeraPerformanceCollectorTest PRIVATE -Wall -Wextra -Werror)
endif()
enable_testing()
add_test(NAME CollectorContract COMMAND ChimeraPerformanceCollectorTest "${CMAKE_CURRENT_BINARY_DIR}/collector-only.csv")

cmake_minimum_required(VERSION 3.22)
project(ChimeraPreparation LANGUAGES CXX)
# Deliberately separate from the production JUCE/plugin build; no downloads.
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(ChimeraBoardStateTests tests/board_state_test.cpp)
target_include_directories(ChimeraBoardStateTests PRIVATE include)
if(MSVC)
  target_compile_options(ChimeraBoardStateTests PRIVATE /W4 /utf-8)
else()
  target_compile_options(ChimeraBoardStateTests PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME FiveSlotState COMMAND ChimeraBoardStateTests)

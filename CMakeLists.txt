cmake_minimum_required(VERSION 3.22)
project(ChimeraAmpMatrix VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.8)
FetchContent_MakeAvailable(JUCE)
set(CHIMERA_FORMATS VST3 Standalone)
if(APPLE) list(APPEND CHIMERA_FORMATS AU) endif()
juce_add_plugin(ChimeraAmpMatrix COMPANY_NAME SpectralForge BUNDLE_ID audio.spectralforge.chimera IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE COPY_PLUGIN_AFTER_BUILD FALSE PLUGIN_MANUFACTURER_CODE SpFg PLUGIN_CODE Chm1 FORMATS ${CHIMERA_FORMATS} PRODUCT_NAME "Chimera Amp Matrix")
target_sources(ChimeraAmpMatrix PRIVATE Source/PluginProcessor.cpp Source/PluginProcessor.h Source/PluginEditor.cpp Source/PluginEditor.h Source/ChimeraDSP.h)
target_compile_definitions(ChimeraAmpMatrix PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(ChimeraAmpMatrix PRIVATE juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
enable_testing()
juce_add_console_app(ChimeraTests PRODUCT_NAME "Chimera Tests")
target_sources(ChimeraTests PRIVATE Tests/ChimeraTests.cpp)
target_include_directories(ChimeraTests PRIVATE Source)
target_link_libraries(ChimeraTests PRIVATE juce::juce_dsp juce::juce_audio_basics juce::juce_audio_formats PUBLIC juce::juce_recommended_config_flags)
add_test(NAME ChimeraTests COMMAND ChimeraTests)
